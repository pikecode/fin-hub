**部署步骤**:
```bash
# 1. 拉取代码
git pull origin main

# 2. 更新依赖
cd apps/api && uv sync
cd ../admin-web && pnpm install

# 3. 运行数据库迁移
cd apps/api && uv run alembic upgrade head

# 4. 配置环境变量
cp .env.example .env
# 编辑 .env 设置生产配置

# 5. 构建和启动
docker compose -f infra/docker/docker-compose.yml up -d

# 6. 配置自动备份
crontab -e
# 添加: 0 3 * * * /app/scripts/backup-database.sh >> /var/log/backup.log 2>&1
```
